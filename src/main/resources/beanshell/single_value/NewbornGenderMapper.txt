import org.slf4j.Logger;

import java.util.List;
import java.util.Map;

public class NewbornGenderMapper {

    private Logger logger;

    public NewbornGenderMapper(Logger logger) { this.logger = logger; }

    public MappingResult doCustomPredictionMapping(Map predictionKeyMap, Long rootPipelineId) {
        String logMsg = "[RootPipelineID: " + rootPipelineId + "] Entered NewbornGenderMapper doCustomPredictionMapping method";
        logger.info(logMsg);

        Object newbornGenderObj = predictionKeyMap.get("newborn_gender");

        if (newbornGenderObj instanceof List) {
            List newbornGenderList = (List) newbornGenderObj;

            for (int i = 0; i < newbornGenderList.size(); i++) {
                Object obj = newbornGenderList.get(i);
                if (obj instanceof in.handyman.raven.lib.model.bsh.PostProcessingExecutorInput) {
                    in.handyman.raven.lib.model.bsh.PostProcessingExecutorInput newbornGenderInput = (in.handyman.raven.lib.model.bsh.PostProcessingExecutorInput) obj;
                    String validatedGenderValue = genderValidation(newbornGenderInput.getExtractedValue(), rootPipelineId);
                    newbornGenderInput.setExtractedValue(validatedGenderValue);
                }
            }
            predictionKeyMap.put("newborn_gender", newbornGenderList);
        }
        return new MappingResult(predictionKeyMap);
    }

    public String genderValidation(String newbornGender, Long rootPipelineId) {
        logger.info("[RootPipelineID: " + rootPipelineId + "] Entered NewbornGenderMapper genderValidation method");
        return normalizeGender(newbornGender);
    }

    public String normalizeGender(String gender) {
        if (gender == null || gender.trim().isEmpty()) {
            return "";
        }
        gender = gender.trim().toLowerCase();

        switch (gender) {
            case "m":
            case "male":
                return "M";
            case "f":
            case "female":
                return "F";
            default:
                return "";
        }
    }

    public static class MappingResult {
        private Map mappedData;

        public MappingResult(Map mappedData) {
            this.mappedData = mappedData != null ? mappedData : new java.util.HashMap();
        }

        public Map getMappedData() {
            return mappedData;
        }
    }
}
