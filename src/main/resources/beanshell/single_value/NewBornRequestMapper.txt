import org.slf4j.Logger;
import java.util.List;
import java.util.Map;

public class NewBornRequestMapper {

    private Logger logger;

    public NewBornRequestMapper(Logger logger) {
        this.logger = logger;
    }

    public MappingResult doCustomPredictionMapping(Map data, Long rootPipelineId) {
        String logMsg = "[RootPipelineID: " + rootPipelineId + "] Entered NewBornRequestMapper doCustomPredictionMapping method";
        logger.info(logMsg);

        if (data == null) {
            return new MappingResult(new java.util.HashMap());
        }

        Object newbornRequestObj = data.get("newborn_request");

        if (newbornRequestObj instanceof List) {
            List newbornRequestList = (List) newbornRequestObj;

            for (int i = 0; i < newbornRequestList.size(); i++) {
                Object obj = newbornRequestList.get(i);
                if (obj instanceof PostProcessingExecutorInput) {
                    PostProcessingExecutorInput newbornRequestInput = (PostProcessingExecutorInput) obj;

                    String promptValue = getExtractedValueFromList(data.get("newborn_request_prompt"), i);
                    String ocrValue = getExtractedValueFromList(data.get("newborn_request_ocr"), i);

                    boolean isPromptY = false;
                    boolean isOcrY = false;

                    if (promptValue != null && !promptValue.trim().equals("")) {
                        isPromptY = "Y".equalsIgnoreCase(promptValue.trim());
                    }
                    if (ocrValue != null && !ocrValue.trim().equals("")) {
                        isOcrY = "Y".equalsIgnoreCase(ocrValue.trim());
                    }

                    String result = (isPromptY && isOcrY) ? "Y" : "";

                    newbornRequestInput.setExtractedValue(result);

                    logger.info("[RootPipelineID: " + rootPipelineId +
                            "] newborn_request index " + i + " mapped to '" + result +
                            "' (prompt: '" + promptValue + "', ocr: '" + ocrValue + "')");
                }
            }
            data.put("newborn_request", newbornRequestList);
        }
        return new MappingResult(data);
    }

    private String getExtractedValueFromList(Object listObj, int index) {
        if (listObj instanceof List) {
            List list = (List) listObj;
            if (index < list.size()) {
                Object item = list.get(index);
                if (item instanceof PostProcessingExecutorInput) {
                    String value = ((PostProcessingExecutorInput) item).getExtractedValue();
                    return (value != null) ? value : "";
                }
            }
        }
        return "";
    }

    public static class MappingResult {
        private Map mappedData;

        public MappingResult(Map mappedData) {
            this.mappedData = mappedData != null ? mappedData : new java.util.HashMap();
        }

        public Map getMappedData() {
            return mappedData;
        }
    }

    public static class PostProcessingExecutorInput {

        private Long tenantId;
        private double aggregatedScore;
        private double maskedScore;
        private String originId;
        private Integer paperNo;
        private String extractedValue;
        private double vqaScore;
        private Integer rank;
        private Integer sorItemAttributionId;
        private String sorItemName;
        private String documentId;
        private Long accTransactionId;
        private String label;
        private String sectionAlias;
        private Long score;
        private String bBox;
        private Long rootPipelineId;
        private Long frequency;
        private Long questionId;
        private Long synonymId;
        private String modelRegistry;
        private String encryptionPolicy;
        private String isEncrypted;
        private String lineItemType;

        public String getExtractedValue() {
            return extractedValue;
        }

        public void setExtractedValue(String extractedValue) {
            this.extractedValue = extractedValue;
        }
    }
}
