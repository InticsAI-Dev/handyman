import org.slf4j.Logger;
import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class MedicaidMemberIdValidator {

    private Long rootPipelineId;

    private static final Pattern UM_PATTERN = Pattern.compile("^[uU][mM]\\d{8}$");
    private static final Pattern UM_INVALID_PATTERN = Pattern.compile("^[uU][mM][0-9]*$");
    private static final Pattern WITH_PARENTHESES = Pattern.compile("^([a-zA-Z0-9]+)(?:\\([^)]*\\))?$");
    private static final Pattern REGEX_ALL_ALPHA_PATTERN = Pattern.compile("^[A-Za-z]+$");

    private Logger logger;
    private List authIdCandidates;
    private List additionalAuthCandidates;

    public MedicaidMemberIdValidator(Logger logger) {
        this.logger = logger;
        this.authIdCandidates = new ArrayList();
        this.additionalAuthCandidates = new ArrayList();
    }

    public MappingResult doCustomPredictionMapping(Map predictionKeyMap, Long rootPipelineId) {
        this.rootPipelineId = rootPipelineId;
        if (logger != null) logger.info("[RootPipelineID: " + rootPipelineId + "] Entered doCustomPredictionMapping method");

        if (predictionKeyMap == null) {
            if (logger != null) logger.error("[RootPipelineID: " + rootPipelineId + "] Input map is null for medicaid member id validator");
            return new MappingResult(predictionKeyMap);
        }

        Map resultMap = new Hashtable(predictionKeyMap);

        // Initialize lists for auth_id and additional_auth_properties if they don't exist or are null
        if (!resultMap.containsKey("auth_id") || resultMap.get("auth_id") == null) {
            resultMap.put("auth_id", new ArrayList());
        }
        if (!resultMap.containsKey("additional_auth_properties") || resultMap.get("additional_auth_properties") == null) {
            resultMap.put("additional_auth_properties", new ArrayList());
        }
        if (!resultMap.containsKey("medicaid_id") || resultMap.get("medicaid_id") == null) {
            resultMap.put("medicaid_id", new ArrayList());
        }

        processField(resultMap, "member_id");
        processField(resultMap, "medicaid_id");

        mapAuthIdAndAdditionalProperties(resultMap, predictionKeyMap);

        if (logger != null) logger.info("[RootPipelineID: " + rootPipelineId + "] Completed doCustomPredictionMapping method");
        return new MappingResult(resultMap);
    }

    private void processField(Map resultMap, String fieldName) {
        Object valueObj = resultMap.get(fieldName);

        if (valueObj instanceof List) {
            List objList = (List) valueObj;

            for (int i = 0; i < objList.size(); i++) {
                Object obj = objList.get(i);
                if (obj instanceof PostProcessingExecutorInput) {
                    PostProcessingExecutorInput input = (PostProcessingExecutorInput) obj;
                    String value = input.getExtractedValue() != null ? input.getExtractedValue().trim() : "";

                    if (value.length() == 0) {
                        input.setExtractedValue("");
                        continue;
                    }

                    // Split multiple IDs on /, :, -
                    String[] parts = value.split("[/:]");
                    if (parts.length > 1) {
                        String firstPart = parts[0] != null ? cleanString(parts[0]) : "";
                        String secondPart = parts[1] != null ? cleanString(parts[1]) : "";

                        boolean firstPartIsAuth = false;
                        boolean secondPartIsAuth = false;

                        // First part check
                        if (firstPart != null && !firstPart.isEmpty() && firstPart.toUpperCase().startsWith("UM")) {
                            if (UM_PATTERN.matcher(firstPart).matches()) {
                                addAuthId(resultMap, input, firstPart);
                                authIdCandidates.add(firstPart);
                                firstPartIsAuth = true;
                            } else if (UM_INVALID_PATTERN.matcher(firstPart).matches()) {
                                addAdditionalAuth(resultMap, input, firstPart);
                                additionalAuthCandidates.add(firstPart);
                                firstPartIsAuth = true;
                            }
                        }

                        // Second part check
                        if (secondPart != null && !secondPart.isEmpty() && secondPart.toUpperCase().startsWith("UM")) {
                            if (UM_PATTERN.matcher(secondPart).matches()) {
                                addAuthId(resultMap, input, secondPart);
                                authIdCandidates.add(secondPart);
                                secondPartIsAuth = true;
                            } else if (UM_INVALID_PATTERN.matcher(secondPart).matches()) {
                                addAdditionalAuth(resultMap, input, secondPart);
                                additionalAuthCandidates.add(secondPart);
                                secondPartIsAuth = true;
                            }
                        }

                        // Update current input based on what was found
                        if (firstPartIsAuth && secondPartIsAuth) {
                            input.setExtractedValue("");
                        } else if (firstPartIsAuth) {
                            // First part moved to auth
                             if (fieldName.equals("member_id")) {
                                 // If member_id, first part is Auth, so member_id is cleared.
                                 // Second part (which is NOT Auth) goes to medicaid_id.
                                 input.setExtractedValue("");
                                 addToMedicaidId(resultMap, input, secondPart);
                             } else {
                                 // If medicaid_id, first part is Auth.
                                 // Second part stays in medicaid_id.
                                 valueCheck(input, secondPart);
                             }
                        } else if (secondPartIsAuth) {
                            // Second part moved to auth, first part remains
                            valueCheck(input, firstPart);
                        } else {
                            // Neither is auth
                            if (fieldName.equals("member_id")) {
                                valueCheck(input, firstPart);
                                // We need to add second part to medicaid_id list
                                addToMedicaidId(resultMap, input, secondPart);
                            } else {
                                // For medicaid_id field
                                valueCheck(input, secondPart);
                            }
                        }

                    } else {
                        // Single value processing
                        value = cleanString(value);

                        if (value != null && !value.isEmpty() && value.toUpperCase().startsWith("UM")) {
                            if (UM_PATTERN.matcher(value).matches()) {
                                addAuthId(resultMap, input, value);
                                authIdCandidates.add(value);
                                input.setExtractedValue("");
                            } else if (UM_INVALID_PATTERN.matcher(value).matches()) {
                                addAdditionalAuth(resultMap, input, value);
                                additionalAuthCandidates.add(value);
                                input.setExtractedValue("");
                            } else {
                                // Starts with UM but not matching patterns? Treat as invalid/empty?
                                // Original code: resultMap.put(fieldName, "");
                                input.setExtractedValue("");
                            }
                        } else {
                            valueCheck(input, value);
                        }
                    }
                }
            }
        }
    }

    private void valueCheck(PostProcessingExecutorInput input, String value) {
        if (value == null || value.isEmpty()) {
            input.setExtractedValue("");
            return;
        }

        Matcher parenMatcher = WITH_PARENTHESES.matcher(value);
        if (parenMatcher.find()) {
            value = parenMatcher.group(1);
            if (value == null) {
                value = "";
            }
        }

        if (value.length() > 20 && !REGEX_ALL_ALPHA_PATTERN.matcher(value).matches()) {
            input.setExtractedValue(value.substring(0, 20));
        }
        else if (REGEX_ALL_ALPHA_PATTERN.matcher(value).matches()) {
            input.setExtractedValue("");
        } else {
            input.setExtractedValue(value);
        }
    }

    private void addAuthId(Map resultMap, PostProcessingExecutorInput originalInput, String value) {
        List authList = (List) resultMap.get("auth_id");
        PostProcessingExecutorInput newInput = copyInput(originalInput);
        newInput.setExtractedValue(value);
        newInput.setSorItemName("auth_id");
        authList.add(newInput);
    }

    private void addAdditionalAuth(Map resultMap, PostProcessingExecutorInput originalInput, String value) {
        List addAuthList = (List) resultMap.get("additional_auth_properties");
        PostProcessingExecutorInput newInput = copyInput(originalInput);
        newInput.setExtractedValue(value);
        newInput.setSorItemName("additional_auth_properties");
        addAuthList.add(newInput);
    }

    private void addToMedicaidId(Map resultMap, PostProcessingExecutorInput originalInput, String value) {
        if (!resultMap.containsKey("medicaid_id") || resultMap.get("medicaid_id") == null) {
            resultMap.put("medicaid_id", new ArrayList());
        }
        List list = (List) resultMap.get("medicaid_id");
        PostProcessingExecutorInput newInput = copyInput(originalInput);
        // Apply value check logic to the new medicaid id
        valueCheck(newInput, value);
        newInput.setSorItemName("medicaid_id");
        list.add(newInput);
    }

    private PostProcessingExecutorInput copyInput(PostProcessingExecutorInput input) {
        if (input == null) {
            return new PostProcessingExecutorInput();
        }

        PostProcessingExecutorInput newInput = new PostProcessingExecutorInput();
        newInput.tenantId = input.tenantId;
        newInput.aggregatedScore = input.aggregatedScore;
        newInput.maskedScore = input.maskedScore;
        newInput.originId = input.originId;
        newInput.paperNo = input.paperNo;
        newInput.vqaScore = input.vqaScore;
        newInput.rank = input.rank;
        newInput.sorItemAttributionId = input.sorItemAttributionId;
        newInput.sorItemName = input.sorItemName;
        newInput.documentId = input.documentId;
        newInput.accTransactionId = input.accTransactionId;
        newInput.label = input.label;
        newInput.sectionAlias = input.sectionAlias;
        newInput.score = input.score;
        newInput.bBox = input.bBox;
        newInput.rootPipelineId = input.rootPipelineId;
        newInput.frequency = input.frequency;
        newInput.questionId = input.questionId;
        newInput.synonymId = input.synonymId;
        newInput.modelRegistry = input.modelRegistry;
        newInput.encryptionPolicy = input.encryptionPolicy;
        newInput.isEncrypted = input.isEncrypted;
        newInput.lineItemType = input.lineItemType;
        return newInput;
    }

    private void mapAuthIdAndAdditionalProperties(Map resultMap, Map inputMap) {
        // This method was originally processing "auth_id" from inputMap.
        // Since we are now building auth_id list dynamically in processField,
        // we might need to process existing auth_id inputs if any.

        Object authIdObj = inputMap.get("auth_id");
        if (authIdObj instanceof List) {
            List authIdList = (List) authIdObj;

            // Let's iterate and validate existing items
            for (int i = 0; i < authIdList.size(); i++) {
                Object obj = authIdList.get(i);
                if (obj instanceof PostProcessingExecutorInput) {
                    PostProcessingExecutorInput input = (PostProcessingExecutorInput) obj;

                    String authId = input.getExtractedValue() != null ? cleanString(input.getExtractedValue()) : null;

                    if (authId != null && authId.length() != 0) {
                        if (UM_PATTERN.matcher(authId).matches()) {
                            authIdCandidates.add(authId);
                            input.setExtractedValue(authId); // Update with cleaned value
                        } else if (UM_INVALID_PATTERN.matcher(authId).matches()) {
                            additionalAuthCandidates.add(authId);
                            // Move to additional_auth_properties
                            addAdditionalAuth(resultMap, input, authId);
                            input.setExtractedValue(""); // Clear from auth_id
                        } else {
                        }
                    }
                }
            }
        }
        List finalAuthList = (List) resultMap.get("auth_id");
        List finalAddAuthList = (List) resultMap.get("additional_auth_properties");

        if (!authIdCandidates.isEmpty()) {

             List validAuths = new ArrayList();
             List invalidAuths = new ArrayList(); // Items that were in auth_id but are actually additional

             // Separate items in finalAuthList
             if (finalAuthList != null) {
                 Iterator it = finalAuthList.iterator();
                 while(it.hasNext()) {
                     Object obj = it.next();
                     if (obj instanceof PostProcessingExecutorInput) {
                         PostProcessingExecutorInput input = (PostProcessingExecutorInput) obj;
                         String val = input.getExtractedValue();
                         if (val != null && !val.isEmpty()) {
                             validAuths.add(input);
                         } else {
                             // Empty value? Remove it?
                             it.remove();
                         }
                     }
                 }
             }

             if (!validAuths.isEmpty()) {
                 // Keep the first one
                 PostProcessingExecutorInput primary = (PostProcessingExecutorInput) validAuths.get(0);

                 // Move others
                 for (int i = 1; i < validAuths.size(); i++) {
                     PostProcessingExecutorInput extra = (PostProcessingExecutorInput) validAuths.get(i);
                     extra.setSorItemName("additional_auth_properties");
                     finalAddAuthList.add(extra);
                 }

                 finalAuthList.clear();
                 finalAuthList.add(primary);
             } else {
                 if (finalAuthList.isEmpty()) {
                 }
             }

        } else {
            finalAuthList.clear();
        }


    }

    private String cleanString(String str) {
        if (str == null || str.trim().length() == 0) return "";
        str = str.trim();
        if (str == null || str.isEmpty()) return "";
        int parantheseIndex = str.indexOf('(');
        if (parantheseIndex != -1) {
            str = str.substring(0, parantheseIndex);
            if (str == null) str = "";
        }
        return str.replaceAll("[^a-zA-Z0-9]", "");
    }

    public static class MappingResult {
        private Map mappedData;

        public MappingResult(Map mappedData) {
            this.mappedData = mappedData != null ? mappedData : new HashMap();
        }

        public Map getMappedData() {
            return mappedData;
        }
    }

    public static class PostProcessingExecutorInput {

        private Long tenantId;
        private double aggregatedScore;
        private double maskedScore;
        private String originId;
        private Integer paperNo;
        private String extractedValue;
        private double vqaScore;
        private Integer rank;
        private Integer sorItemAttributionId;
        private String sorItemName;
        private String documentId;
        private Long accTransactionId;
        private String label;
        private String sectionAlias;
        private Long score;
        private String bBox;
        private Long rootPipelineId;
        private Long frequency;
        private Long questionId;
        private Long synonymId;
        private String modelRegistry;
        private String encryptionPolicy;
        private String isEncrypted;
        private String lineItemType;

        public String getExtractedValue() {
            return extractedValue;
        }

        public void setExtractedValue(String extractedValue) {
            this.extractedValue = extractedValue;
        }
        public void setSorItemName(String sorItemName) {
            this.sorItemName = sorItemName;
        }

    }
}
