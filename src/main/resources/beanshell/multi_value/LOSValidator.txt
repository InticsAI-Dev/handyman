import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.*;

public class LOSValidator {

    private static Set URGENT_KEYWORDS;
    private static Set LOS_KEYWORDS;
    private static Set BLACKLIST_KEYWORDS;

    static {
        URGENT_KEYWORDS = new HashSet();
        URGENT_KEYWORDS.add("asap");
        URGENT_KEYWORDS.add("urgent");
        URGENT_KEYWORDS.add("rush");
        URGENT_KEYWORDS.add("expedite");
        URGENT_KEYWORDS.add("stat");
        URGENT_KEYWORDS.add("2nd request");
        URGENT_KEYWORDS.add("second request");
        URGENT_KEYWORDS.add("emergency");
        URGENT_KEYWORDS.add("er");
        URGENT_KEYWORDS.add("expedited");

        LOS_KEYWORDS = new HashSet();
        LOS_KEYWORDS.add("notice of admission");
        LOS_KEYWORDS.add("notification of admission");

        BLACKLIST_KEYWORDS = new HashSet();
        BLACKLIST_KEYWORDS.add("dont rush me");
        BLACKLIST_KEYWORDS.add("urgently");
        BLACKLIST_KEYWORDS.add("rushing");
        BLACKLIST_KEYWORDS.add("non-urgent");
        BLACKLIST_KEYWORDS.add("non urgent");
        BLACKLIST_KEYWORDS.add("emergency dept");
        BLACKLIST_KEYWORDS.add("emergency contact");
        BLACKLIST_KEYWORDS.add("emergency medicine");
        BLACKLIST_KEYWORDS.add("emergency department");
        BLACKLIST_KEYWORDS.add("non-emergency");
        BLACKLIST_KEYWORDS.add("non emergency");
        BLACKLIST_KEYWORDS.add("non-emergent");
        BLACKLIST_KEYWORDS.add("non emergent");
        BLACKLIST_KEYWORDS.add("non-expeditiously");
        BLACKLIST_KEYWORDS.add("non expeditiously");
        BLACKLIST_KEYWORDS.add("don't rush me");
    }

    private Logger logger;

    public LOSValidator(Logger logger) {
        this.logger = (logger != null) ? logger : LoggerFactory.getLogger(LOSValidator.class);
    }

    public MappingResult doCustomPredictionMapping(Map data, Long rootPipelineId) {
        logInfo(rootPipelineId, "Starting Level of Service validation process");

        if (data == null) {
            logInfo(rootPipelineId, "Input map is null");
            return new MappingResult(data);
        }

        processLevelOfService(data, rootPipelineId);
        return new MappingResult(data);
    }

    private void processLevelOfService(Map data, Long rootPipelineId) {
        Object levelOfServiceObj = data.get("level_of_service");

        if (!(levelOfServiceObj instanceof List)) {
            logWarn(rootPipelineId, "level_of_service is not a List: " + levelOfServiceObj);
            return;
        }

        List levelOfServiceList = (List) levelOfServiceObj;
        if (levelOfServiceList.isEmpty()) {
            logInfo(rootPipelineId, "Empty level_of_service list");
            return;
        }

        for (Object obj : levelOfServiceList) {
            if (!(obj instanceof PostProcessingExecutorInput)) {
                logWarn(rootPipelineId, "level_of_service list contains non-PostProcessingExecutorInput object: " + obj);
                continue;
            }

            PostProcessingExecutorInput input = (PostProcessingExecutorInput) obj;
            String extractedValue = input.getExtractedValue();

            if (extractedValue == null || extractedValue.trim().isEmpty()) {
                input.setExtractedValue("");
                input.setLabel("");
                input.setSectionAlias("");
                input.setBBox("");
                continue;
            }

            String levelOfService = extractedValue.trim();
            String levelOfServicetoLower = levelOfService.replaceAll("\\s+", " ").toLowerCase();
            boolean isUrgent = false;
            boolean blacklistFound = false;
            for (Object keyword : BLACKLIST_KEYWORDS) {
                if (levelOfServicetoLower.contains((String) keyword)) {
                    blacklistFound = true;
                    logInfo(rootPipelineId, "Blacklist keyword found: " + keyword + ", setting all level_of_service to empty");
                    break;
                }
            }
            if (!blacklistFound && levelOfServicetoLower.matches("\\b(er|ER)(\\s|$|-).*")) {
                isUrgent = true;
                logInfo(rootPipelineId,
                        "Valid level_of_service, setting to Urgent due to exact match");
            } else {
                for (Object keyword : URGENT_KEYWORDS) {
                    if (!keyword.equals("er") && !blacklistFound
                            && levelOfServicetoLower.contains((String) keyword)) {
                        isUrgent = true;
                        logInfo(rootPipelineId,
                                "Valid level_of_service, setting to Urgent due to keyword match");
                        break;
                    }
                }
            }

            if (!isUrgent && !blacklistFound) {
                for (Object keyword : LOS_KEYWORDS) {
                    if (levelOfServicetoLower.contains((String) keyword)) {
                        boolean hasOneAuthId = hasOneAuthIdValue(data, rootPipelineId);
                        if (hasOneAuthId) {
                            isUrgent = false;
                            logInfo(rootPipelineId,
                                    "Valid level_of_service, UM Id has one value. Setting Urgent = false");
                        } else {
                            isUrgent = true;
                            logInfo(rootPipelineId,
                                    "Valid level_of_service, UM Id does not have one value. Setting Urgent = true");
                        }
                        break;
                    }
                }
            }

            String finalValue = isUrgent ? "Urgent" : "";
            input.setExtractedValue(finalValue);
            if (finalValue.isEmpty()) {
                input.setLabel("");
                input.setSectionAlias("");
                input.setBBox("");
            }
        }

        data.put("level_of_service", levelOfServiceList);
    }

    private boolean hasOneAuthIdValue(Map data, Long rootPipelineId) {
        Object authIdObj = data.get("auth_id");
        if (!(authIdObj instanceof List)) {
            return false;
        }

        List authIdList = (List) authIdObj;
        int validCount = 0;
        for (Object obj : authIdList) {
            if (!(obj instanceof PostProcessingExecutorInput)) {
                logWarn(rootPipelineId, "auth_id list contains non-PostProcessingExecutorInput object: " + obj);
                continue;
            }

            PostProcessingExecutorInput input = (PostProcessingExecutorInput) obj;
            String value = input.getExtractedValue();
            if (value != null && !value.trim().isEmpty()) {
                validCount++;
            }
        }
        return validCount != 0;
    }


    private void logInfo(Long rootPipelineId, String message) {
        logger.info("[RootPipelineID: " + rootPipelineId + "] " + message);
    }

    private void logWarn(Long rootPipelineId, String message) {
        logger.warn("[RootPipelineID: " + rootPipelineId + "] " + message);
    }

    public static class MappingResult {
        private Map mappedData;

        public MappingResult(Map mappedData) {
            this.mappedData = (mappedData != null) ? new HashMap(mappedData) : new HashMap();
        }

        public Map getMappedData() {
            return new HashMap(mappedData);
        }
    }

    public static class PostProcessingExecutorInput {

        private Long tenantId;
        private double aggregatedScore;
        private double maskedScore;
        private String originId;
        private Integer paperNo;
        private String extractedValue;
        private double vqaScore;
        private Integer rank;
        private Integer sorItemAttributionId;
        private String sorItemName;
        private String documentId;
        private Long accTransactionId;
        private String label;
        private String sectionAlias;
        private Long score;
        private String bBox;
        private Long rootPipelineId;
        private Long frequency;
        private Long questionId;
        private Long synonymId;
        private String modelRegistry;
        private String encryptionPolicy;
        private String isEncrypted;
        private String lineItemType;

        public String getExtractedValue() {
            return extractedValue;
        }

        public void setExtractedValue(String extractedValue) {
            this.extractedValue = extractedValue;
        }

        public void setLabel(String label) { this.label = label; }

        public void setSectionAlias(String sectionAlias) { this.sectionAlias = sectionAlias; }

        public void setBBox(String bBox) { this.bBox = bBox; }

        public String getLabel() { return label; }

        public String getSectionAlias() { return sectionAlias; }

        public String getBBox() { return bBox; }

    }
}
